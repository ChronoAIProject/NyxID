import os,signal,subprocess,sys,time,shutil,json,ctypes,struct,resource
from pathlib import Path
def process_argv(pid):
 libc=ctypes.CDLL(None,use_errno=True)
 mib=(ctypes.c_int*3)(1,49,pid) # macOS KERN_PROCARGS2: read argc argv only, never env.
 size=ctypes.c_size_t()
 if libc.sysctl(mib,3,None,ctypes.byref(size),None,0)!=0:return None
 buffer=ctypes.create_string_buffer(size.value)
 if libc.sysctl(mib,3,buffer,ctypes.byref(size),None,0)!=0:return None
 raw=buffer.raw[:size.value]
 argc=struct.unpack_from('i',raw)[0]
 offset=raw.index(b'\0',4)+1
 while offset<len(raw) and raw[offset]==0:offset+=1
 return [part.decode('utf-8','replace') for part in raw[offset:].split(b'\0')[:argc]]
def observe_owned_argv(group,observed,output):
 rows=subprocess.run(['ps','-axo','pid=,pgid=,comm='],capture_output=True,text=True,check=True).stdout
 for row in rows.splitlines():
  fields=row.split(None,2)
  if len(fields)!=3 or int(fields[1])!=group:continue
  name=Path(fields[2]).name
  if name not in ('cargo','cargo-clippy','rustc','clippy-driver'):continue
  argv=process_argv(int(fields[0]))
  if not argv:continue
  if name in ('rustc','clippy-driver') and not ('--crate-name' in argv and argv[argv.index('--crate-name')+1]=='nyxid_server'):continue
  identity=(int(fields[0]),tuple(argv))
  if identity in observed:continue
  observed.add(identity)
  output.write(json.dumps({'at_UTC':time.strftime('%Y-%m-%dT%H:%M:%SZ',time.gmtime()),'pid':int(fields[0]),'pgid':group,'name':name,'argv':argv})+'\n');output.flush()
resource.setrlimit(resource.RLIMIT_NOFILE,(65536,resource.getrlimit(resource.RLIMIT_NOFILE)[1]))
assert resource.getrlimit(resource.RLIMIT_NOFILE)[0]==65536
label=sys.argv[1]
command=sys.argv[2:]
resource_config='/Users/chronoai/Library/Application Support/heca/worktrees/eafaaf47/arctic-eagle/docs/plans/2026-10-27-nyxid-workstreams/cargo-resource-profile.toml'
if command and Path(command[0]).name=='cargo':
 if command[1:2]==['--config']:
  assert command[2:3]==[resource_config], 'Cargo must use the coordinator resource config'
  del command[1:3]
 if command[1:2]==['clippy']:
  # cargo-clippy must pass --config through to its backend Cargo child.
  if command[2:3]==['--config']:
   assert command[3:4]==[resource_config], 'Clippy must use the coordinator resource config'
  else:
   command[2:2]=['--config',resource_config]
 else:
  command[1:1]=['--config',resource_config]
assert label and all(c.isalnum() or c=='-' for c in label)
base=Path('/tmp')/('nyxid-permissions-r3-'+label)
env=os.environ.copy()
env.update(CARGO_TARGET_DIR='/Users/chronoai/Desktop/projects/nyxid/target',CARGO_BUILD_JOBS='2',NYXID_TEST_DATABASE_URL='mongodb://127.0.0.1:27471/?replicaSet=nyxidPermissionsR3')
free=lambda:shutil.disk_usage('/Users/chronoai/Desktop/projects/nyxid/target').free/2**30
start_free=free()
if start_free<2:
 print('Refused owned command before launch: free_GiB=%.3f'%start_free,flush=True)
 raise SystemExit(2)
start=time.monotonic()
print('command:', ' '.join(command), 'start_free_GiB=%.3f'%start_free,flush=True)
minimum=start_free
with base.with_suffix('.log').open('w') as log,base.with_suffix('.disk.log').open('w') as disk,base.with_suffix('.argv.jsonl').open('w') as arguments:
 proc=subprocess.Popen(command,env=env,stdout=log,stderr=subprocess.STDOUT,start_new_session=True)
 print('owned_process_group:',proc.pid,flush=True)
 observed=set()
 i=0
 stopped=False
 while proc.poll() is None:
  observe_owned_argv(proc.pid,observed,arguments)
  available=free();minimum=min(minimum,available)
  if i%5==0:
   disk.write(time.strftime('%Y-%m-%dT%H:%M:%SZ',time.gmtime())+' free_GiB=%.3f\n'%available);disk.flush()
  if available<2:
   stopped=True
   disk.write('STOP: below 2 GiB; terminating owned process group %s\n'%proc.pid);disk.flush()
   os.killpg(proc.pid,signal.SIGTERM)
   try:proc.wait(timeout=5)
   except subprocess.TimeoutExpired:os.killpg(proc.pid,signal.SIGKILL);proc.wait(timeout=5)
   # Also stop a child that survived the parent's exit in this owned group.
   try:os.killpg(proc.pid,signal.SIGKILL)
   except ProcessLookupError:pass
   children=subprocess.run(['pgrep','-g',str(proc.pid)],capture_output=True,text=True).stdout.strip()
   disk.write('owned group survivors after stop: '+(children or 'none')+'\n');disk.flush()
   break
  i+=1;time.sleep(0.05 if i<20 else 3)
 rc=proc.wait()
survivors=subprocess.run(['pgrep','-g',str(proc.pid)],capture_output=True,text=True).stdout.strip()
end_free=free();minimum=min(minimum,end_free)
result={'command':command,'owned_process_group':proc.pid,'owned_group_survivors':survivors.splitlines(),'nofile_soft':resource.getrlimit(resource.RLIMIT_NOFILE)[0],'cwd':os.getcwd(),'exit_code':rc,'resource_stop':stopped,'elapsed_seconds':round(time.monotonic()-start,2),'start_free_GiB':round(start_free,3),'min_free_GiB':round(minimum,3),'end_free_GiB':round(end_free,3),'log':str(base.with_suffix('.log')),'argv_log':str(base.with_suffix('.argv.jsonl'))}
base.with_suffix('.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps(result),flush=True)
raise SystemExit(rc if rc>=0 else 128-rc)
