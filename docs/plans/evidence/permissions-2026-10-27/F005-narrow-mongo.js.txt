let step = 0;
function requireTrue(value) { ++step; if (!value) throw new Error("Mongo readiness assertion failed at step " + step); }
function requireEqual(actual, expected) { requireTrue(actual === expected); }
function requireWorked(value) { requireTrue(value && (value.ok === 1 || value.acknowledged === true)); }
const admin = db.getSiblingDB("admin");
const hello = admin.runCommand({hello:1});
requireWorked(hello);
requireEqual(hello.setName, "nyxidPermissionsR3");
requireEqual(hello.isWritablePrimary, true);
requireEqual(hello.me, "127.0.0.1:27471");
requireEqual(hello.hosts.length, 1);
requireEqual(hello.hosts[0], "127.0.0.1:27471");
requireTrue(hello.logicalSessionTimeoutMinutes > 0);
const build = admin.runCommand({buildInfo:1});
requireWorked(build);
requireEqual(build.versionArray[0], 8);
const opts = admin.runCommand({getCmdLineOpts:1});
requireWorked(opts);
requireEqual(opts.parsed.net.port, 27471);
requireEqual(opts.parsed.net.bindIp, "127.0.0.1");
requireEqual(opts.parsed.storage.dbPath, "/tmp/nyxid-permissions-r3-mongo/data");
const fixture = db.getSiblingDB("nyxidPermissionsR3_F005_narrow_preflight");
const name = "probe_" + new ObjectId().toHexString();
requireWorked(fixture.createCollection(name));
const session = db.getMongo().startSession();
const collection = session.getDatabase(fixture.getName()).getCollection(name);
session.startTransaction({readConcern:{level:"snapshot"},writeConcern:{w:"majority"}});
requireWorked(collection.insertOne({_id:"committed",ready:true}));
session.commitTransaction();
requireEqual(fixture.getCollection(name).countDocuments({_id:"committed",ready:true}), 1);
session.startTransaction();
requireWorked(collection.insertOne({_id:"aborted",ready:true}));
session.abortTransaction();
requireEqual(fixture.getCollection(name).countDocuments({_id:"aborted"}), 0);
session.endSession();
print(JSON.stringify({version:build.version,setName:hello.setName,host:hello.me,primary:hello.isWritablePrimary,dbPath:opts.parsed.storage.dbPath,commitVerified:true,abortVerified:true,probeDatabase:fixture.getName(),probeCollection:name,retainedDataPreserved:true}));
